cmake_minimum_required(VERSION 3.16)
project(krita_mcp_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Xml)
find_package(KF6I18n REQUIRED)
find_path(BOOST_HEADERS boost/operators.hpp HINTS "${CMAKE_CURRENT_SOURCE_DIR}/build/deps/usr/include")
if(NOT BOOST_HEADERS)
  message(FATAL_ERROR "Boost headers are required (libboost-dev / boost package)")
endif()
find_library(KRITA_COMMAND kritacommand REQUIRED)
find_path(EIGEN_HEADERS Eigen/Sparse PATH_SUFFIXES eigen3 HINTS "${CMAKE_CURRENT_SOURCE_DIR}/build/deps/usr/include")
if(NOT EIGEN_HEADERS)
  message(FATAL_ERROR "Eigen headers are required (libeigen3-dev / eigen package)")
endif()
find_library(KRITA_UI kritaui REQUIRED)
add_library(krita_mcp_native SHARED bridge.cpp)
target_compile_options(krita_mcp_native PRIVATE -fno-operator-names)
file(GLOB KF_INCLUDE_ROOTS "/usr/include/KF6/*")
foreach(KF_PATH IN LISTS KF_INCLUDE_ROOTS)
  if(NOT IS_DIRECTORY "${KF_PATH}")
    list(REMOVE_ITEM KF_INCLUDE_ROOTS "${KF_PATH}")
  endif()
endforeach()
target_include_directories(krita_mcp_native PRIVATE vendor build/krita_headers ${BOOST_HEADERS} ${EIGEN_HEADERS} ${KF_INCLUDE_ROOTS})
find_library(KRITA_LIBKIS kritalibkis REQUIRED)
find_library(KRITA_IMAGE kritaimage REQUIRED)
target_link_libraries(krita_mcp_native PRIVATE Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Xml KF6::I18n ${KRITA_COMMAND} ${KRITA_UI} ${KRITA_LIBKIS} ${KRITA_IMAGE})
